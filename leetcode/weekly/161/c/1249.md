对于一个右括号 $s[i]$，我们需要找到 $s[i]$ 左边最近未匹配的左括号，与之匹配。如果没有左括号，那么 $s[i]$ 无法匹配，必须删除。例如 $\texttt{())}$ 中的第二个右括号必须删除。

用栈记录未匹配的左括号的下标。遍历到右括号 $s[i]$ 时，如果栈非空，则弹出栈顶，否则标记 $s[i]$ 需要删除。

遍历完 $s$ 后，如果栈非空，那么栈中这些左括号都没有与之匹配的右括号，标记为待删除。

代码实现时，可以先把待删除的括号改成 $\texttt{-}$ 号（或者 `'\0'`），最后只保留非 $\texttt{-}$ 号字符，即为答案。Python 可以直接改成空。

```py [sol-Python3]
class Solution:
    def minRemoveToMakeValid(self, s: str) -> str:
        t = list(s)
        st = []

        for i, ch in enumerate(t):
            if ch == '(':
                st.append(i)  # 记录左括号的下标
            elif ch == ')':
                if st:
                    st.pop()  # 左右括号匹配
                else:
                    t[i] = ''  # 右括号没有与之匹配的左括号，标记为移除

        for i in st:
            t[i] = ''  # 栈中剩下的左括号没有与之匹配的右括号，标记为移除

        return ''.join(t)
```

```java [sol-Java]
class Solution {
    public String minRemoveToMakeValid(String s) {
        char[] t = s.toCharArray();
        int[] st = new int[t.length]; // 用数组模拟栈
        int size = 0;

        for (int i = 0; i < t.length; i++) {
            if (t[i] == '(') {
                st[size++] = i; // 记录左括号的下标
            } else if (t[i] == ')') {
                if (size > 0) {
                    size--; // 左右括号匹配
                } else {
                    t[i] = 0; // 右括号没有与之匹配的左括号，标记为移除
                }
            }
        }

        for (int i = 0; i < size; i++) {
            t[st[i]] = 0; // 栈中剩下的左括号没有与之匹配的右括号，标记为移除
        }

        // 不需要创建 StringBuilder，可以直接把非零元素移到 t 的前面
        // 原理见 283. 移动零 https://leetcode.cn/problems/move-zeroes/
        size = 0;
        for (char ch : t) {
            if (ch != 0) {
                t[size++] = ch;
            }
        }
        return new String(t, 0, size);
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    string minRemoveToMakeValid(string s) {
        vector<int> st;

        for (int i = 0; i < s.size(); i++) {
            if (s[i] == '(') {
                st.push_back(i); // 记录左括号的下标
            } else if (s[i] == ')') {
                if (!st.empty()) {
                    st.pop_back(); // 左右括号匹配
                } else {
                    s[i] = 0; // 右括号没有与之匹配的左括号，标记为移除
                }
            }
        }

        for (int i : st) {
            s[i] = 0; // 栈中剩下的左括号没有与之匹配的右括号，标记为移除
        }

        string ans;
        for (char ch : s) {
            if (ch) {
                ans += ch;
            }
        }
        return ans;
    }
};
```

```go [sol-Go]
func minRemoveToMakeValid(s string) string {
	t := []byte(s)
	st := []int{}

	for i, ch := range t {
		if ch == '(' {
			st = append(st, i) // 记录左括号的下标
		} else if ch == ')' {
			if len(st) > 0 {
				st = st[:len(st)-1] // 左右括号匹配
			} else {
				t[i] = '-' // 右括号没有与之匹配的左括号，标记为移除
			}
		}
	}

	for _, i := range st {
		t[i] = '-' // 栈中剩下的左括号没有与之匹配的右括号，标记为移除
	}

	return strings.Replace(string(t), "-", "", -1)
}
```
