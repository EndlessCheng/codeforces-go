把下标 $0$ 到 $|s|-1$ 当作节点，$\textit{pairs}$ 当作边，创建一个无向图 $G$。节点 $v$ 的点权为 $s_v$。

**定理**：$G$ 的同一个连通块中的点权可以**随意交换**。

[证明](https://zhuanlan.zhihu.com/p/2027293665970721660)。

## 写法一：建图 + 排序

```py [sol-Python3]
class Solution:
    def smallestStringWithSwaps(self, s: str, pairs: List[List[int]]) -> str:
        n = len(s)
        g = [[] for _ in range(n)]
        for x, y in pairs:
            g[x].append(y)
            g[y].append(x)
        vis = [False] * n

        def dfs(x: int) -> None:
            vis[x] = True
            idx.append(x)  # 收集这个连通块的下标
            for y in g[x]:
                if not vis[y]:
                    dfs(y)

        ans = [''] * n
        for start, done in enumerate(vis):
            if done:
                continue

            idx = []
            dfs(start)
            idx.sort()  # 下标从小到大排序

            t = sorted(s[i] for i in idx)  # 字母从小到大排序

            # 小的字母填入小的下标，大的字母填入大的下标
            for i, ch in zip(idx, t):
                ans[i] = ch

        return ''.join(ans)
```

```java [sol-Java]
class Solution {
    public String smallestStringWithSwaps(String S, List<List<Integer>> pairs) {
        char[] s = S.toCharArray();
        int n = s.length;
        List<Integer>[] g = new ArrayList[n];
        Arrays.setAll(g, _ -> new ArrayList<>());
        for (List<Integer> p : pairs) {
            int x = p.get(0);
            int y = p.get(1);
            g[x].add(y);
            g[y].add(x);
        }

        boolean[] vis = new boolean[n];
        List<Integer> idx = new ArrayList<>();

        for (int start = 0; start < n; start++) {
            if (vis[start]) {
                continue;
            }

            idx.clear();
            dfs(start, g, vis, idx);
            Collections.sort(idx); // 下标从小到大排序

            int m = idx.size();
            char[] t = new char[m];
            for (int i = 0; i < m; i++) {
                t[i] = s[idx.get(i)];
            }
            Arrays.sort(t); // 字母从小到大排序

            // 小的字母填入小的下标，大的字母填入大的下标
            for (int i = 0; i < m; i++) {
                s[idx.get(i)] = t[i];
            }
        }

        return new String(s);
    }

    private void dfs(int x, List<Integer>[] g, boolean[] vis, List<Integer> idx) {
        vis[x] = true;
        idx.add(x); // 收集这个连通块的下标
        for (int y : g[x]) {
            if (!vis[y]) {
                dfs(y, g, vis, idx);
            }
        }
    }
}
```

```go [sol-Go]
func smallestStringWithSwaps(s string, pairs [][]int) string {
	n := len(s)
	g := make([][]int, n)
	for _, p := range pairs {
		x, y := p[0], p[1]
		g[x] = append(g[x], y)
		g[y] = append(g[y], x)
	}

	vis := make([]bool, n)
	idx := []int{}

	var dfs func(int)
	dfs = func(x int) {
		vis[x] = true
		idx = append(idx, x) // 收集这个连通块的下标
		for _, y := range g[x] {
			if !vis[y] {
				dfs(y)
			}
		}
	}

	ans := make([]byte, n)
	for start, b := range vis {
		if b {
			continue
		}

		idx = idx[:0]
		dfs(start)
		slices.Sort(idx) // 下标从小到大排序

		t := make([]byte, len(idx))
		for i, j := range idx {
			t[i] = s[j]
		}
		slices.Sort(t) // 字母从小到大排序

		// 小的字母填入小的下标，大的字母填入大的下标
		for i, ch := range t {
			ans[idx[i]] = ch
		}
	}

	return string(ans)
}
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(m + n\log n)$，其中 $n$ 是 $s$ 的长度，$m$ 是 $\textit{pairs}$ 的长度。
- 空间复杂度：$\mathcal{O}(n + m)$。

## 写法二：并查集 + 排序

```py [sol-Python3]
class Solution:
    def smallestStringWithSwaps(self, s: str, pairs: List[List[int]]) -> str:
        n = len(s)
        # 初始化并查集
        fa = list(range(n))  # fa[i] = i

        def find(x: int) -> int:
            if fa[x] != x:
                fa[x] = find(fa[x])
            return fa[x]

        # 合并
        for x, y in pairs:
            fa[find(x)] = find(y)

        # 把同一个连通块中的下标分到同一组
        idx_groups = defaultdict(list)
        for i in range(n):  # i 从小到大，所以 list 保存的下标已经有序，后面无需排序
            idx_groups[find(i)].append(i)

        ans = [''] * n
        for idx in idx_groups.values():
            t = sorted(s[i] for i in idx)  # 字母从小到大排序
            # 小的字母填入小的下标，大的字母填入大的下标
            for i, ch in zip(idx, t):
                ans[i] = ch

        return ''.join(ans)
```

```java [sol-Java]
class Solution {
    public String smallestStringWithSwaps(String S, List<List<Integer>> pairs) {
        char[] s = S.toCharArray();
        int n = s.length;
        // 初始化并查集
        int[] fa = new int[n];
        Arrays.setAll(fa, i -> i);

        // 合并
        for (List<Integer> p : pairs) {
            int x = find(p.get(0), fa);
            int y = find(p.get(1), fa);
            fa[x] = y;
        }

        // 把同一个连通块中的下标分到同一组
        List<Integer>[] idxGroups = new ArrayList[n];
        Arrays.setAll(idxGroups, _ -> new ArrayList<>());
        for (int i = 0; i < n; i++) { // i 从小到大，所以保存的下标已经有序，后面无需排序
            int rt = find(i, fa);
            idxGroups[rt].add(i);
        }

        for (List<Integer> idx : idxGroups) {
            int m = idx.size();
            char[] t = new char[m];
            for (int i = 0; i < m; i++) {
                t[i] = s[idx.get(i)];
            }
            Arrays.sort(t); // 字母从小到大排序

            // 小的字母填入小的下标，大的字母填入大的下标
            for (int i = 0; i < m; i++) {
                s[idx.get(i)] = t[i];
            }
        }

        return new String(s);
    }

    private int find(int x, int[] fa) {
        if (fa[x] != x) {
            fa[x] = find(fa[x], fa);
        }
        return fa[x];
    }
}
```
