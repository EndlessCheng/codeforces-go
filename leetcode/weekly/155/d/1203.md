## 核心思路

题目要求「同一小组的项目，排序后在列表中彼此相邻」，说明一个组可以看成一个整体，所以可以先确定**组与组之间的顺序**，再确定**组内的项目顺序**。

这都可以用**拓扑排序**解决。

## 具体思路

例如项目 $2$ 依赖项目 $1$，而项目 $1$ 属于组 $1$，项目 $2$ 属于组 $2$，所以组 $1$ 必须排在组 $2$ 之前。

一般地，如果项目 $x$ 依赖项目 $y = \textit{beforeItems}[x]$：

- 如果 $x$ 和 $y$ 属于不同组，那么在「组图」$\textit{groupG}$ 中添加有向边 $\textit{group}[y] \to \textit{group}[x]$。
- 如果 $x$ 和 $y$ 属于同一组，那么在「项目图」$\textit{itemG}$ 中添加有向边 $y \to x$。

建图后，在 $\textit{groupG}$ 上跑拓扑排序，得到组与组之间的顺序 $\textit{groupOrder}$。然后遍历 $\textit{groupOrder}$，对每个组，在 $\textit{itemG}$（的节点子集）上跑拓扑排序，得到组内的项目顺序。

为了知道每个组包含哪些项目（$\textit{itemG}$ 的节点子集），可以先遍历 $\textit{group}$，把同一组的项目收集到同一个列表中。

对于不属于任何小组的项目，为了方便拓扑排序，可以对这些项目分别单独创建新的小组，并分配小组编号，每个组只包含一个项目。

```py [sol-Python3]
# 模板来源：https://leetcode.cn/discuss/post/3581143/
# 返回有向图 g 的节点子集 sub_nodes 的拓扑序
# 如果图中有环，返回 None
def topologicalSort(g: list[list[int]], in_deg: list[int], sub_nodes: list[int]) -> list[int]:
    q = deque(x for x in sub_nodes if in_deg[x] == 0)
    topo_order = []

    while q:
        x = q.popleft()
        topo_order.append(x)
        # 节点 y 依赖节点 x
        for y in g[x]:
            in_deg[y] -= 1  # y 的依赖数量减一
            if in_deg[y] == 0:  # y 的依赖都排在 y 的左边了，可以排 y
                q.append(y)

    if len(topo_order) < len(sub_nodes):  # 图中有环
        return None
    return topo_order


class Solution:
    def sortItems(self, n: int, m: int, group: list[int], beforeItems: list[list[int]]) -> list[int]:
        # groupItems[i] 保存组 i 的所有项目
        group_items = [[] for _ in range(m + n)]
        for i, g in enumerate(group):
            if g < 0:  # 项目 i 不属于任何小组
                g = group[i] = m  # 创建一个新的小组，方便拓扑排序
                m += 1
            group_items[g].append(i)

        # 建图，顺带计算入度
        group_g = [[] for _ in range(m)]
        group_in_deg = [0] * m
        item_g = [[] for _ in range(n)]
        item_in_deg = [0] * n
        for x, items in enumerate(beforeItems):
            gx = group[x]
            for y in items:
                gy = group[y]
                if gx != gy:
                    group_g[gy].append(gx)  # 组与组的依赖关系
                    group_in_deg[gx] += 1
                else:
                    item_g[y].append(x)  # 同组项目的依赖关系
                    item_in_deg[x] += 1

        # 计算组与组之间的顺序
        group_order = topologicalSort(group_g, group_in_deg, list(range(m)))
        if group_order is None:  # 循环依赖
            return []

        # 对每个组，计算组内的项目顺序
        ans = []
        for i in group_order:
            item_order = topologicalSort(item_g, item_in_deg, group_items[i])
            if item_order is None:  # 循环依赖
                return []
            ans += item_order
        return ans
```

```java [sol-Java]
class Solution {
    public int[] sortItems(int n, int m, int[] group, List<List<Integer>> beforeItems) {
        // groupItems[i] 保存组 i 的所有项目
        List<Integer>[] groupItems = new ArrayList[m + n];
        Arrays.setAll(groupItems, _ -> new ArrayList<>());
        for (int i = 0; i < n; i++) {
            int g = group[i];
            if (g < 0) { // 项目 i 不属于任何小组
                g = group[i] = m++; // 创建一个新的小组，方便拓扑排序
            }
            groupItems[g].add(i);
        }

        // 建图，顺带计算入度
        List<Integer>[] groupG = new ArrayList[m];
        Arrays.setAll(groupG, _ -> new ArrayList<>());
        int[] groupInDeg = new int[m];
        List<Integer>[] itemG = new ArrayList[n];
        Arrays.setAll(itemG, _ -> new ArrayList<>());
        int[] itemInDeg = new int[n];
        for (int x = 0; x < n; x++) {
            int gx = group[x];
            for (int y : beforeItems.get(x)) {
                int gy = group[y];
                if (gx != gy) {
                    groupG[gy].add(gx); // 组与组的依赖关系
                    groupInDeg[gx]++;
                } else {
                    itemG[y].add(x); // 同组项目的依赖关系
                    itemInDeg[x]++;
                }
            }
        }

        // 计算组与组之间的顺序
        List<Integer> groupIds = new ArrayList<>(m);
        for (int i = 0; i < m; i++) {
            groupIds.add(i);
        }
        int[] groupOrder = topologicalSort(groupG, groupInDeg, groupIds);
        if (groupOrder == null) { // 循环依赖
            return new int[0];
        }

        // 对每个组，计算组内的项目顺序
        int[] ans = new int[n];
        int size = 0;
        for (int i : groupOrder) {
            int[] itemOrder = topologicalSort(itemG, itemInDeg, groupItems[i]);
            if (itemOrder == null) { // 循环依赖
                return new int[0];
            }
            System.arraycopy(itemOrder, 0, ans, size, itemOrder.length);
            size += itemOrder.length;
        }
        return ans;
    }

    // 模板来源：https://leetcode.cn/discuss/post/3581143/
    // 返回有向图 g 的节点子集 subNodes 的拓扑序
    // 如果图中有环，返回空列表
    public int[] topologicalSort(List<Integer>[] g, int[] inDeg, List<Integer> subNodes) {
        Queue<Integer> q = new ArrayDeque<>();
        for (int x : subNodes) {
            if (inDeg[x] == 0) { // 节点 x 没有依赖，加入拓扑序
                q.offer(x);
            }
        }

        int[] topoOrder = new int[subNodes.size()];
        int size = 0;
        while (!q.isEmpty()) {
            int x = q.poll();
            topoOrder[size++] = x;
            // 节点 y 依赖节点 x
            for (int y : g[x]) {
                inDeg[y]--; // y 的依赖数量减一
                if (inDeg[y] == 0) { // y 的依赖都排在 y 的左边了，可以排 y
                    q.offer(y);
                }
            }
        }

        if (size < subNodes.size()) { // 图中有环
            return null;
        }
        return topoOrder;
    }
}
```

```cpp [sol-C++]
class Solution {
    // 模板来源：https://leetcode.cn/discuss/post/3581143/
    // 返回有向图 g 的节点子集 sub_nodes 的拓扑序
    // 如果图中有环，返回空
    vector<int> topologicalSort(vector<vector<int>>& g, vector<int>& in_deg, vector<int>& sub_nodes) {
        queue<int> q;
        for (int x : sub_nodes) {
            if (in_deg[x] == 0) { // 节点 x 没有依赖，加入拓扑序
                q.push(x);
            }
        }

        vector<int> topo_order;
        while (!q.empty()) {
            int x = q.front();
            q.pop();
            topo_order.push_back(x);
            // 节点 y 依赖节点 x
            for (int y : g[x]) {
                in_deg[y]--; // y 的依赖数量减一
                if (in_deg[y] == 0) { // y 的依赖都排在 y 的左边了，可以排 y
                    q.push(y);
                }
            }
        }

        if (topo_order.size() < sub_nodes.size()) { // 图中有环
            return {};
        }
        return topo_order;
    }

public:
    vector<int> sortItems(int n, int m, vector<int>& group, vector<vector<int>>& beforeItems) {
        // groupItems[i] 保存组 i 的所有项目
        vector<vector<int>> group_items(m + n);
        for (int i = 0; i < group.size(); i++) {
            int& g = group[i];
            if (g < 0) { // 项目 i 不属于任何小组
                g = m++; // 创建一个新的小组，方便拓扑排序
            }
            group_items[g].push_back(i);
        }

        // 建图，顺带计算入度
        vector<vector<int>> group_g(m), item_g(n);
        vector<int> group_in_deg(m), item_in_deg(n);
        for (int x = 0; x < beforeItems.size(); x++) {
            int gx = group[x];
            for (int y : beforeItems[x]) {
                int gy = group[y];
                if (gx != gy) {
                    group_g[gy].push_back(gx); // 组与组的依赖关系
                    group_in_deg[gx]++;
                } else {
                    item_g[y].push_back(x); // 同组项目的依赖关系
                    item_in_deg[x]++;
                }
            }
        }

        // 计算组与组之间的顺序
        vector<int> group_ids(m);
        ranges::iota(group_ids, 0);
        auto group_order = topologicalSort(group_g, group_in_deg, group_ids);
        if (group_order.empty()) { // 循环依赖
            return {};
        }

        // 对每个组，计算组内的项目顺序
        vector<int> ans;
        for (int i : group_order) {
            auto& sub_nodes = group_items[i];
            if (sub_nodes.empty()) {
                continue;
            }
            auto item_order = topologicalSort(item_g, item_in_deg, sub_nodes);
            if (item_order.empty()) { // 循环依赖
                return {};
            }
            ans.insert(ans.end(), item_order.begin(), item_order.end());
        }
        return ans;
    }
};
```

```go [sol-Go]
// 模板来源：https://leetcode.cn/discuss/post/3581143/
// 返回有向图 g 的节点子集 subNodes 的拓扑序
// 如果图中有环，返回 nil
func topologicalSort(g [][]int, inDeg []int, subNodes []int) []int {
	n := len(subNodes)
	q := make([]int, 0, n)
	topoOrder := q
	for _, x := range subNodes {
		if inDeg[x] == 0 { // 节点 x 没有依赖，加入拓扑序
			q = append(q, x)
		}
	}

	for len(q) > 0 {
		x := q[0]
		q = q[1:]
		// 节点 y 依赖节点 x
		for _, y := range g[x] {
			inDeg[y]-- // y 的依赖数量减一
			if inDeg[y] == 0 { // y 的依赖都排在 y 的左边了，可以排 y
				q = append(q, y)
			}
		}
	}

	if cap(q) > 0 { // 有节点没入队，说明图中有环
		return nil
	}
	return topoOrder[:n]
}

func sortItems(n int, m int, group []int, beforeItems [][]int) (ans []int) {
	// groupItems[i] 保存组 i 的所有项目
	groupItems := make([][]int, m+n)
	for i, g := range group {
		if g < 0 { // 项目 i 不属于任何小组
			group[i] = m // 创建一个新的小组，方便拓扑排序
			g = m
			m++
		}
		groupItems[g] = append(groupItems[g], i)
	}

	// 建图，顺带计算入度
	groupG := make([][]int, m)
	groupInDeg := make([]int, m)
	itemG := make([][]int, n)
	itemInDeg := make([]int, n)
	for x, items := range beforeItems {
		gx := group[x]
		for _, y := range items {
			gy := group[y]
			if gx != gy {
				groupG[gy] = append(groupG[gy], gx) // 组与组的依赖关系
				groupInDeg[gx]++
			} else {
				itemG[y] = append(itemG[y], x) // 同组项目的依赖关系
				itemInDeg[x]++
			}
		}
	}

	// 计算组与组之间的顺序
	groupIDs := make([]int, m)
	for i := range groupIDs {
		groupIDs[i] = i
	}
	groupOrder := topologicalSort(groupG, groupInDeg, groupIDs)
	if groupOrder == nil { // 循环依赖
		return
	}

	// 对每个组，计算组内的项目顺序
	for _, i := range groupOrder {
		itemOrder := topologicalSort(itemG, itemInDeg, groupItems[i])
		if itemOrder == nil { // 循环依赖
			return nil
		}
		ans = append(ans, itemOrder...)
	}
	return
}
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n+m+E)$，其中 $E$ 是所有 $\textit{beforeItems}[i]$ 的长度之和。
- 空间复杂度：$\mathcal{O}(n+m+E)$。
