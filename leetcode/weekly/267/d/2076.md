## 题意

有一个 $n$ 个点的无向图，一开始没有边。遍历 $\textit{requests}$，把 $\textit{requests}[i]$ 这条边添加到图中，但如果添加这条边会导致节点 $\textit{restrictions}[j][0]$ 和 $\textit{restrictions}[j][1]$ 在同一个连通块，那么不能添加这条边。

## 分析

设添加的边为 $u\text{-}v$，设 $x = \textit{restrictions}[j][0],\ y = \textit{restrictions}[j][1]$。

分类讨论：

- 如果 $x$ 和 $u$ 在同一个连通块，且 $y$ 和 $v$ 在同一个连通块，那么连边会导致 $x$ 和 $y$ 在同一个连通块，请求失败。
- 如果 $x$ 和 $v$ 在同一个连通块，且 $y$ 和 $u$ 在同一个连通块，那么连边会导致 $x$ 和 $y$ 在同一个连通块，请求失败。
- 其余情况，请求成功。

暴力枚举所有 $\textit{restrictions}[j]$，用**并查集**快速判断两个点是否在同一个连通块中。

```py [sol-Python3]
class Solution:
    def friendRequests(self, n: int, restrictions: List[List[int]], requests: List[List[int]]) -> List[bool]:
        # 初始化并查集
        fa = list(range(n))  # fa[i] = i

        def find(x: int) -> int:
            if fa[x] != x:
                fa[x] = find(fa[x])
            return fa[x]

        ans = [False] * len(requests)
        for i, (u, v) in enumerate(requests):
            u, v = find(u), find(v)
            # 暴力检查
            for x, y in restrictions:
                x, y = find(x), find(y)
                if x == u and y == v or x == v and y == u:
                    break  # 不能连边
            else:
                # 可以连边
                fa[u] = v  # 并查集合并
                ans[i] = True
        return ans
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n + mk\log n)$，其中 $m$ 是 $\textit{requests}$ 的长度，$k$ 是 $\textit{restrictions}$ 的长度。只有路径压缩的并查集是 $\mathcal{O}(\log n)$ 的。
- 空间复杂度：$\mathcal{O}(n)$。返回值不计入。

## 优化：启发式合并

举个例子，假设用户 $1$ 不能和用户 $3,4,7$ 成为朋友，用户 $2$ 不能和用户 $5,6,7$ 成为朋友。那么当用户 $1$ 和用户 $2$ 成为朋友后，$1$ 和 $2$ 所在的连通块不能和用户 $3,4,5,6,7$ 成为朋友。

一般地，定义 $\textit{ban}[i][x]$ 表示代表元为 $i$ 的连通块不能与代表元为 $x$ 的连通块合并。

当我们把代表元为 $u$ 的连通块与代表元为 $v$ 的连通块合并时，把 $\textit{ban}[u]$ 并入 $\textit{ban}[v]$，表示无法与 $u$ 合并的连通块，也无法与 $v$ 合并。

代码实现时，并查集使用按大小合并，并且在合并 $\textit{ban}$ 时，把连通块小的一方合并到连通块大的一方（启发式合并），从而减少时间复杂度，详见复杂度分析。

```py [sol-Python3]
class Solution:
    def friendRequests(self, n: int, restrictions: List[List[int]], requests: List[List[int]]) -> List[bool]:
        # 初始化并查集
        fa = list(range(n))  # fa[i] = i
        size = [1] * n

        def find(x: int) -> int:
            if fa[x] != x:
                fa[x] = find(fa[x])
            return fa[x]

        # ban[i][x] 表示代表元为 i 的连通块不能与代表元为 x 的连通块合并
        ban = [set() for _ in range(n)]
        # 初始化无法合并的连通块
        for x, y in restrictions:
            ban[x].add(y)
            ban[y].add(x)  # 双向

        ans = [False] * len(requests)
        for i, (u, v) in enumerate(requests):
            # 把代表元为 u 的连通块与代表元为 v 的连通块合并
            u, v = find(u), find(v)
            if u == v:  # 已经合并
                ans[i] = True
                continue
            if v in ban[u]:  # 无法合并
                continue

            # 启发式合并：总是把小的连通块合并到大的连通块
            if size[u] > size[v]:
                u, v = v, u

            # 把 u 并入 v，那么无法与 u 合并的连通块，也无法与 v 合并
            for x in ban[u]:
                ban[v].add(x)
                ban[x].discard(u)  # 删掉旧的
                ban[x].add(v)  # 添加新的
            ban[u] = None  # 保证空间复杂度

            fa[u] = v  # 并查集合并
            size[v] += size[u]
            ans[i] = True

        return ans
```

```java [sol-Java]
class Solution {
    public boolean[] friendRequests(int n, int[][] restrictions, int[][] requests) {
        // 初始化并查集
        int[] fa = new int[n];
        Arrays.setAll(fa, i -> i); // fa[i] = i
        int[] size = new int[n];
        Arrays.fill(size, 1);

        // ban[i][x] 表示代表元为 i 的连通块不能与代表元为 x 的连通块合并
        Set<Integer>[] ban = new HashSet[n];
        Arrays.setAll(ban, _ -> new HashSet<>());
        // 初始化无法合并的连通块
        for (int[] r : restrictions) {
            int x = r[0];
            int y = r[1];
            ban[x].add(y);
            ban[y].add(x); // 双向
        }

        boolean[] ans = new boolean[requests.length];
        for (int i = 0; i < requests.length; i++) {
            // 把代表元为 u 的连通块与代表元为 v 的连通块合并
            int u = find(requests[i][0], fa);
            int v = find(requests[i][1], fa);
            if (u == v) { // 已经合并
                ans[i] = true;
                continue;
            }
            if (ban[u].contains(v)) { // 无法合并
                continue;
            }

            // 启发式合并：总是把小的连通块合并到大的连通块
            if (size[u] > size[v]) {
                // 交换 u 和 v
                int tmp = u;
                u = v;
                v = tmp;
            }

            // 把 u 并入 v，那么无法与 u 合并的连通块，也无法与 v 合并
            for (int x : ban[u]) {
                ban[v].add(x);
                ban[x].remove(u); // 删掉旧的
                ban[x].add(v); // 添加新的
            }
            ban[u] = null; // 保证空间复杂度

            fa[u] = v; // 并查集合并
            size[v] += size[u];
            ans[i] = true;
        }
        return ans;
    }

    private int find(int x, int[] fa) {
        if (fa[x] != x) {
            fa[x] = find(fa[x], fa);
        }
        return fa[x];
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    vector<bool> friendRequests(int n, vector<vector<int>>& restrictions, vector<vector<int>>& requests) {
        // 初始化并查集
        vector<int> fa(n);
        ranges::iota(fa, 0); // fa[i] = i;
        vector<int> size(n, 1);

        auto find = [&](this auto&& find, int x) -> int {
            if (fa[x] != x) {
                fa[x] = find(fa[x]);
            }
            return fa[x];
        };

        // ban[i][x] 表示代表元为 i 的连通块不能与代表元为 x 的连通块合并
        vector<unordered_set<int>> ban(n);
        // 初始化无法合并的连通块
        for (auto& r : restrictions) {
            int x = r[0], y = r[1];
            ban[x].insert(y);
            ban[y].insert(x);
        }

        vector<bool> ans(requests.size());
        for (int i = 0; i < requests.size(); i++) {
            // 把代表元为 u 的连通块与代表元为 v 的连通块合并
            int u = find(requests[i][0]);
            int v = find(requests[i][1]);
            if (u == v) { // 已经合并
                ans[i] = true;
                continue;
            }
            if (ban[u].contains(v)) { // 无法合并
                continue;
            }

            // 启发式合并：总是把小的连通块合并到大的连通块
            if (size[u] > size[v]) {
                swap(u, v);
            }

            // 把 u 并入 v，那么无法与 u 合并的连通块，也无法与 v 合并
            for (int x : ban[u]) {
                ban[v].insert(x);
                ban[x].erase(u); // 删掉旧的
                ban[x].insert(v); // 添加新的
            }
            ban[u].clear(); // 保证空间复杂度

            fa[u] = v; // 并查集合并
            size[v] += size[u];
            ans[i] = true;
        }
        return ans;
    }
};
```

```go [sol-Go]
func friendRequests(n int, restrictions [][]int, requests [][]int) []bool {
	// 初始化并查集
	fa := make([]int, n)
	size := make([]int, n)
	for i := range fa {
		fa[i] = i
		size[i] = 1
	}
	var find func(int) int
	find = func(x int) int {
		if fa[x] != x {
			fa[x] = find(fa[x])
		}
		return fa[x]
	}

	// ban[i][x] 表示代表元为 i 的连通块不能与代表元为 x 的连通块合并
	ban := make([]map[int]bool, n)
	for i := range ban {
		ban[i] = map[int]bool{}
	}
	// 初始化无法合并的连通块
	for _, r := range restrictions {
		x, y := r[0], r[1]
		ban[x][y] = true
		ban[y][x] = true // 双向
	}

	ans := make([]bool, len(requests))
	for i, r := range requests {
		// 把代表元为 u 的连通块与代表元为 v 的连通块合并
		u, v := find(r[0]), find(r[1])
		if u == v { // 已经合并
			ans[i] = true
			continue
		}
		if ban[u][v] { // 无法合并
			continue
		}

		// 启发式合并：总是把小的连通块合并到大的连通块
		if size[u] > size[v] {
			u, v = v, u
		}

		// 把 u 并入 v，那么无法与 u 合并的连通块，也无法与 v 合并
		for x := range ban[u] {
			ban[v][x] = true
			delete(ban[x], u) // 删掉旧的
			ban[x][v] = true // 添加新的
		}
		ban[u] = nil // 保证空间复杂度

		fa[u] = v // 并查集合并
		size[v] += size[u]
		ans[i] = true
	}
	return ans
}
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n + m\alpha(n) + k\log n)$，其中 $m$ 是 $\textit{requests}$ 的长度，$k$ 是 $\textit{restrictions}$ 的长度。并查集同时使用了路径压缩和按大小合并，时间复杂度为 $\mathcal{O}(n + m\alpha(n))$。下面分析二重循环的时间复杂度。每当 $u$ 所在的连通块合并到更大的连通块时，$\textit{ban}[u]$ 中的元素会被迁移。而每次迁移后，$u$ 所在的连通块的大小至少翻倍。由于连通块大小最多是 $n$，所以最多翻倍 $\mathcal{O}(\log n)$ 次，$\textit{ban}[u]$ 中的每个元素最多迁移 $\mathcal{O}(\log n)$ 次。所以 $k$ 个限制条件一共最多迁移 $\mathcal{O}(k\log n)$ 次。
- 空间复杂度：$\mathcal{O}(n+k)$。返回值不计入。在合并过程中我们清理了不再用到的数据，$\textit{ban}$ 记录的限制数不超过 $2k$。
