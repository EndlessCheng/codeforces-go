本题要处理表达式之间的两种运算：

- 并列（并集）。
- 相接（笛卡尔积）。

表达式展开后，我们得到的是字符串的集合。

对于并列，计算的是两个或多个集合的并集。例如 $\texttt{\{\{a,b\},\{b,c\}\}}$，我们要计算集合 $S = \{\texttt{a},\texttt{b}\}$ 与集合 $T = \{\texttt{b},\texttt{c}\}$ 的并集，即 $S\cup T = \{\texttt{a},\texttt{b},\texttt{c}\}$。

对于相接，以 $\texttt{\{a,b\}\{c,d\}e}$ 为例：

1. 分成三个部分 $\texttt{\{a,b\}}$、$\texttt{\{c,d\}}$ 和 $\texttt{e}$。
2. 第一个集合是 $S = \{\texttt{a},\texttt{b}\}$。
3. 计算 $S = \{\texttt{a},\texttt{b}\}$ 与下一个集合 $T = \{\texttt{c},\texttt{d}\}$ 的笛卡尔积，即 $\{s+t \mid s\in S \wedge t \in T\}$，得到新的集合 $\{\texttt{ac},\texttt{ad},\texttt{bc},\texttt{bd}\}$，令 $S$ 为新的集合。
4. 按照第三步的方法，计算 $S$ 与集合 $T = \{\texttt{e}\}$ 的笛卡尔积，得到新的集合 $\{\texttt{ace},\texttt{ade},\texttt{bce},\texttt{bde}\}$，令 $S$ 为新的集合。
5. 最后 $S$ 即为花括号展开后的字符串集合。

处理表达式时，注意运算的优先级：**并列像加法，相接像乘法**。实现时，我们要准备两个哈希集合：

- 哈希集合 $\textit{res}$，保存整个表达式展开后的所有字符串。
- 哈希集合 $\textit{cur}$，保存当前正在处理的相接表达式的字符串集合。为方便实现，可以初始化 $\textit{cur}$ 为包含一个空串的集合，这样无需特判 $\textit{cur}$ 为空的情况。

表达式有嵌套关系，这可以递归处理。在递归过程中，用一个在递归函数外的变量 $i$ 表示当前下标。每遍历到一个字符，就把 $i$ 加一。

- 遇到字母：把字母加到 $\textit{cur}$ 的每个字符串的末尾。
- 遇到逗号：把 $\textit{cur}$ 中的所有字符串添加到 $\textit{res}$ 中，然后清空 $\textit{cur}$。
- 遇到左括号：递归处理。递归结束后，计算 $\textit{cur}$ 与递归返回值的笛卡尔积。
- 遇到右括号，或者遍历到 $\textit{expression}$ 的末尾：递归结束，先把 $\textit{cur}$ 中的所有字符串添加到 $\textit{res}$ 中，然后返回 $\textit{res}$。

## 写法一：递归

```py [sol-Python3]
class Solution:
    def braceExpansionII(self, expression: str) -> list[str]:
        i = 0

        def dfs() -> set[str]:
            nonlocal i
            res = set()
            cur = {""}  # 加个空串，简化后续判断逻辑

            while i < len(expression):
                ch = expression[i]
                i += 1

                if ch == '}':  # 归
                    break
                if ch == ',':  # 取并集
                    res |= cur  # 把 cur 中的字符串都添加到 res 中
                    cur = {""}
                elif ch == '{':  # 递
                    sub_res = dfs()
                    cur = {s + t for s in cur for t in sub_res}  # 计算 cur 和 sub_res 的笛卡尔积
                else:  # ch 是字母
                    cur = {s + ch for s in cur}  # 把 ch 添加到 cur 每个字符串的末尾

            return res | cur

        return sorted(dfs())
```

```java [sol-Java]
class Solution {
    private int i = 0;

    public List<String> braceExpansionII(String expression) {
        char[] s = expression.toCharArray();
        List<String> ans = new ArrayList<>(dfs(s));
        Collections.sort(ans);
        return ans;
    }

    private Set<String> dfs(char[] expression) {
        Set<String> res = new HashSet<>();
        Set<String> cur = new HashSet<>();
        cur.add(""); // 加个空串，简化后续判断逻辑

        while (i < expression.length) {
            char ch = expression[i];
            i++;

            if (ch == '}') { // 归
                break;
            }
            if (ch == ',') { // 取并集
                res.addAll(cur); // 把 cur 中的字符串都添加到 res 中
                cur.clear();
                cur.add("");
            } else if (ch == '{') { // 递
                Set<String> subRes = dfs(expression);
                // 计算 cur 和 subRes 的笛卡尔积
                Set<String> newSet = new HashSet<>();
                for (String s : cur) {
                    for (String t : subRes) {
                        newSet.add(s + t);
                    }
                }
                cur = newSet;
            } else { // ch 是字母
                Set<String> newSet = new HashSet<>();
                for (String s : cur) {
                    newSet.add(s + ch); // 把 ch 添加到 cur 每个字符串的末尾
                }
                cur = newSet;
            }
        }

        res.addAll(cur);
        return res;
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    vector<string> braceExpansionII(string expression) {
        int i = 0;

        auto dfs = [&](this auto&& dfs) -> unordered_set<string> {
            unordered_set<string> res;
            unordered_set<string> cur = {""}; // 加个空串，简化后续判断逻辑

            while (i < expression.size()) {
                char ch = expression[i];
                i++;

                if (ch == '}') { // 归
                    break;
                }
                if (ch == ',') { // 取并集
                    res.insert(cur.begin(), cur.end()); // 把 cur 中的字符串都添加到 res 中
                    cur = {""};
                } else if (ch == '{') { // 递
                    auto sub_res = dfs();
                    // 计算 cur 和 sub_res 的笛卡尔积
                    unordered_set<string> new_set;
                    for (auto& s : cur) {
                        for (auto& t : sub_res) {
                            new_set.insert(s + t);
                        }
                    }
                    cur = move(new_set);
                } else { // ch 是字母
                    unordered_set<string> new_set;
                    for (auto& s : cur) {
                        new_set.insert(s + ch); // 把 ch 添加到 cur 每个字符串的末尾
                    }
                    cur = move(new_set);
                }
            }

            res.insert(cur.begin(), cur.end());
            return res;
        };

        auto res = dfs();
        vector<string> ans(res.begin(), res.end());
        ranges::sort(ans);
        return ans;
    }
};
```

```go [sol-Go]
func braceExpansionII(expression string) []string {
	type set map[string]struct{}
	i := 0

	var dfs func() set
	dfs = func() set {
		res := set{}
		cur := set{"": {}} // 加个空串，简化后续判断逻辑

		for i < len(expression) {
			ch := expression[i]
			i++

			if ch == '}' { // 归
				break
			}
			if ch == ',' { // 取并集
				maps.Copy(res, cur) // 把 cur 中的字符串都添加到 res 中
				cur = set{"": {}}
			} else if ch == '{' { // 递
				subRes := dfs()
				// 计算 cur 和 subRes 的笛卡尔积
				newSet := make(set, len(cur)*len(subRes)) // 预分配空间
				for s := range cur {
					for t := range subRes {
						newSet[s+t] = struct{}{}
					}
				}
				cur = newSet
			} else { // ch 是字母
				newSet := make(set, len(cur)) // 预分配空间
				t := string(ch)
				for s := range cur {
					newSet[s+t] = struct{}{} // 把 ch 添加到 cur 每个字符串的末尾
				}
				cur = newSet
			}
		}

		maps.Copy(res, cur)
		return res
	}

	return slices.Sorted(maps.Keys(dfs()))
}
```

## 写法二：用栈模拟递归

把递归改写成非递归写法。

你需要理解计算机底层是如何实现递归的，请看视频 [深刻理解递归【基础算法精讲 09】](https://www.bilibili.com/video/BV1UD4y1Y769/)。

简单来说，在往下「递」的时候，计算机会把当前函数中的局部变量保存到一个栈中（这个栈由计算机管理）；下层递归函数「归」之后，再从栈中把保存的变量取出。我们可以手动模拟这个过程，具体实现如下。

```py [sol-Python3]
class Solution:
    def braceExpansionII(self, expression: str) -> list[str]:
        stack = []  # 用于模拟计算机的递归
        res = set()
        cur = {""}  # 加个空串，简化后续判断逻辑

        for ch in expression:
            if ch.isalpha():  # 字母
                cur = {s + ch for s in cur}  # 把 ch 添加到 cur 每个字符串的末尾
            elif ch == ',':  # 取并集
                res |= cur  # 把 cur 中的字符串都添加到 res 中
                cur = {""}
            elif ch == '{':  # 递
                # 模拟递归
                stack.append((res, cur))  # 递归前，把局部变量 res 和 cur 保存到栈中
                res = set()  # 递归，初始化 res 和 cur
                cur = {""}
            else:  # 归
                sub_res = res | cur  # 递归结束，返回值为 sub_res
                res, cur = stack.pop()  # 从栈中恢复递归之前保存的局部变量
                cur = {s + t for s in cur for t in sub_res}  # 计算 cur 和 sub_res 的笛卡尔积

        return sorted(res | cur)
```

```java [sol-Java]
class Solution {
    public List<String> braceExpansionII(String expression) {
        List<Set<String>[]> stack = new ArrayList<>();
        Set<String> res = new HashSet<>();
        Set<String> cur = new HashSet<>();
        cur.add(""); // 加个空串，简化后续判断逻辑

        for (char ch : expression.toCharArray()) {
            if (Character.isLowerCase(ch)) { // 字母
                Set<String> newSet = new HashSet<>();
                for (String s : cur) {
                    newSet.add(s + ch); // 把 ch 添加到 cur 每个字符串的末尾
                }
                cur = newSet;
            } else if (ch == ',') { // 取并集
                res.addAll(cur); // 把 cur 中的字符串都添加到 res 中
                cur.clear();
                cur.add("");
            } else if (ch == '{') { // 递
                // 模拟递归
                stack.add(new Set[]{res, cur}); // 递归前，把局部变量 res 和 cur 保存到栈中
                res = new HashSet<>(); // 递归，初始化 res 和 cur
                cur = new HashSet<>();
                cur.add("");
            } else { // 归
                res.addAll(cur);
                Set<String> subRes = res; // 递归结束，返回值为 subRes

                // 从栈中恢复递归之前保存的局部变量
                Set<String>[] p = stack.removeLast();
                res = p[0];
                cur = p[1];

                // 计算 cur 和 subRes 的笛卡尔积
                Set<String> newSet = new HashSet<>();
                for (String s : cur) {
                    for (String t : subRes) {
                        newSet.add(s + t);
                    }
                }
                cur = newSet;
            }
        }

        res.addAll(cur);
        List<String> ans = new ArrayList<>(res);
        Collections.sort(ans);
        return ans;
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    vector<string> braceExpansionII(string expression) {
        using Set = unordered_set<string>;
        stack<pair<Set, Set>> stk;
        Set res;
        Set cur = {""}; // 加个空串，简化后续判断逻辑

        for (char ch : expression) {
            if (islower(ch)) { // 字母
                Set new_set;
                for (auto& s : cur) {
                    new_set.insert(s + ch); // 把 ch 添加到 cur 每个字符串的末尾
                }
                cur = move(new_set);
            } else if (ch == ',') { // 取并集
                res.insert(cur.begin(), cur.end()); // 把 cur 中的字符串都添加到 res 中
                cur = {""};
            } else if (ch == '{') { // 递
                // 模拟递归
                stk.emplace(move(res), move(cur)); // 递归前，把局部变量 res 和 cur 保存到栈中
                res = {}; // 递归，初始化 res 和 cur
                cur = {""};
            } else { // 归
                res.insert(cur.begin(), cur.end());
                auto sub_res = move(res); // 递归结束，返回值为 sub_res

                // 从栈中恢复递归之前保存的局部变量
                res = move(stk.top().first);
                cur = move(stk.top().second);
                stk.pop();

                // 计算 cur 和 sub_res 的笛卡尔积
                Set new_set;
                for (auto& s : cur) {
                    for (auto& t : sub_res) {
                        new_set.insert(s + t);
                    }
                }
                cur = move(new_set);
            }
        }

        res.insert(cur.begin(), cur.end());
        vector<string> ans(res.begin(), res.end());
        ranges::sort(ans);
        return ans;
    }
};
```

```go [sol-Go]
func braceExpansionII(expression string) []string {
	type set map[string]struct{}
	type pair struct{ res, cur set }

	stack := []pair{}
	res := set{}
	cur := set{"": {}} // 加个空串，简化后续判断逻辑

	for _, ch := range expression {
		if unicode.IsLower(ch) { // 字母
			newSet := make(set, len(cur)) // 预分配空间
			t := string(ch)
			for s := range cur {
				newSet[s+t] = struct{}{} // 把 ch 添加到 cur 每个字符串的末尾
			}
			cur = newSet
		} else if ch == ',' { // 取并集
			maps.Copy(res, cur) // 把 cur 中的字符串都添加到 res 中
			cur = set{"": {}}
		} else if ch == '{' { // 递
			// 模拟递归
			stack = append(stack, pair{res, cur}) // 递归前，把局部变量 res 和 cur 保存到栈中
			res = set{} // 递归，初始化 res 和 cur
			cur = set{"": {}}
		} else { // 归
			maps.Copy(res, cur)
			subRes := res // 递归结束，返回值为 subRes

			// 从栈中恢复递归之前保存的局部变量
			last := len(stack) - 1
			res = stack[last].res
			cur = stack[last].cur
			stack = stack[:last]

			// 计算 cur 和 subRes 的笛卡尔积
			newSet := make(set, len(cur)*len(subRes)) // 预分配空间
			for s := range cur {
				for t := range subRes {
					newSet[s+t] = struct{}{}
				}
			}
			cur = newSet
		}
	}

	maps.Copy(res, cur)
	return slices.Sorted(maps.Keys(res))
}
```

#### 复杂度分析

我们最多能得到多少个不同的字符串？

试想一下，构造很多个相接的表达式，例如 $\textit{expression} = \texttt{\{a,b\}\{a,b\}\{a,b\}}\cdots$ 展开后，字符串的个数为 $2\times 2\times 2\times\cdots$ 是指数级别的。

设 $n$ 是 $\textit{expression}$ 的长度。每个长为 $2k+1$ 的子表达式会贡献 $k$ 的乘积，问题相当于：

- 把一个正整数 $n$ 拆分成若干个奇数 $x_i = 2k_i+1\ (k_i\ge 1)$ 之和，计算 $k_i$ 的乘积的最大值。

根据基本不等式，把 $n$ 均匀分，乘积最大。设分出了 $\dfrac{n}{x}$ 个 $x$，每个数对于乘积的贡献为 $\dfrac{x-1}{2}$，总乘积为

$$
\left(\dfrac{x-1}{2}\right)^{n/x}
$$

求导可知，上式在 $x\approx 7.36$ 处取到最大值。

代入邻近的奇数 $x=7$ 和 $x=9$，由于 $3^{1/7} > 4^{1/9}$，所以当 $x=7$ 时可以得到最大乘积，也就是每个子表达式包含 $3$ 个字母。

所以最多可以得到 $\mathcal{O}(3^{n/7})$ 个不同的字符串。

- 时间复杂度：$\mathcal{O}(n^2 3^{n/7})$，其中 $n$ 是 $\textit{expression}$ 的长度。瓶颈在排序上，需要 $\mathcal{O}(3^{n/7}\log (3^{n/7})) = \mathcal{O}(n 3^{n/7})$ 次比较，每次比两个长为 $\mathcal{O}(n)$ 的字符串需要 $\mathcal{O}(n)$ 的时间，所以排序的时间复杂度为 $\mathcal{O}(n^2 3^{n/7})$。
- 空间复杂度：$\mathcal{O}(3^{n/7})$。
