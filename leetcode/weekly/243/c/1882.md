本题需要维护两类数据：

1. 空闲服务器。我们需要快速找到所有空闲服务器中权重最小的那台，如果有多台权重相同的服务器，取其中下标最小的服务器。这可以用一个**最小堆** $\textit{idle}$ 维护，堆中保存 pair $(服务器权重,服务器下标)$。
2. 繁忙服务器。设当前时刻为 $\textit{now}$，我们需要快速找到所有完成任务的时刻 $\le \textit{now}$ 的繁忙服务器，把这些服务器移至 $\textit{idle}$。这可以用一个**最小堆** $\textit{busy}$ 维护，堆中保存 pair $(服务器完成任务的时刻,服务器下标)$。

一开始 $\textit{now}=0$。我们依次处理每个任务，处理 $\textit{tasks}[i]$ 的流程如下：

1. 更新当前时刻为 $\max(\textit{now}, i)$。
2. 释放 $\textit{busy}$ 中的完成任务的时刻 $\le \textit{now}$ 的所有服务器，移至 $\textit{idle}$。
3. 如果 $\textit{idle}$ 仍然为空，那么**把当前时刻跳至 $\textit{busy}$ 的堆顶时刻**，重复第二步。这可以保证我们有空闲服务器。
4. 弹出 $\textit{idle}$ 的堆顶服务器，处理 $\textit{tasks}[i]$。该任务的完成时刻为 $\textit{now} + \textit{tasks}[i]$，把 $(\textit{now} + \textit{tasks}[i], i)$ 加到 $\textit{busy}$ 中。

```py [sol-Python3]
class Solution:
    def assignTasks(self, servers: List[int], tasks: List[int]) -> List[int]:
        # 初始化空闲服务器堆，保存 (服务器权重, 服务器下标)
        idle = [(w, i) for i, w in enumerate(servers)]
        heapify(idle)

        busy = []  # 繁忙服务器堆，保存 (服务器完成任务的时刻, 服务器下标)
        now = 0  # 当前时刻

        # 把已经完成任务的服务器从 busy 移至 idle
        def release() -> None:
            while busy and busy[0][0] <= now:
                _, i = heappop(busy)
                heappush(idle, (servers[i], i))

        # 依次处理每个任务
        for i, t in enumerate(tasks):
            now = max(now, i)
            release()
            if not idle:
                now = busy[0][0]  # 跳到 busy 的堆顶时刻，从而保证我们有空闲服务器
                release()
            _, j = heappop(idle)
            heappush(busy, (now + t, j))
            tasks[i] = j  # 直接把答案记在 tasks 中
        return tasks
```

```java [sol-Java]
class Solution {
    public int[] assignTasks(int[] servers, int[] tasks) {
        // 初始化空闲服务器堆，保存 (服务器权重, 服务器下标)
        PriorityQueue<int[]> idle = new PriorityQueue<>((a, b) ->
            a[0] != b[0] ? a[0] - b[0] : a[1] - b[1]
        );
        for (int i = 0; i < servers.length; i++) {
            idle.offer(new int[]{servers[i], i});
        }

        // 繁忙服务器堆，保存 (服务器完成任务的时刻, 服务器下标)
        PriorityQueue<long[]> busy = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
        long now = 0; // 当前时刻

        // 依次处理每个任务
        for (int i = 0; i < tasks.length; i++) {
            now = Math.max(now, i);
            release(busy, idle, servers, now);
            if (idle.isEmpty()) {
                now = busy.peek()[0]; // 跳到 busy 的堆顶时刻，从而保证我们有空闲服务器
                release(busy, idle, servers, now);
            }
            int[] p = idle.poll();
            busy.offer(new long[]{now + tasks[i], p[1]});
            tasks[i] = p[1]; // 直接把答案记在 tasks 中
        }
        return tasks;
    }

    // 把已经完成任务的服务器从 busy 移至 idle
    private void release(PriorityQueue<long[]> busy, PriorityQueue<int[]> idle, int[] servers, long now) {
        while (!busy.isEmpty() && busy.peek()[0] <= now) {
            int i = (int) busy.poll()[1];
            idle.offer(new int[]{servers[i], i});
        }
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    vector<int> assignTasks(vector<int>& servers, vector<int>& tasks) {
        // 初始化空闲服务器堆，保存 (服务器权重, 服务器下标)
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> idle;
        for (int i = 0; i < servers.size(); i++) {
            idle.emplace(servers[i], i);
        }

        // 繁忙服务器堆，保存 (服务器完成任务的时刻, 服务器下标)
        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<>> busy;
        long long now = 0; // 当前时刻

        // 把已经完成任务的服务器从 busy 移至 idle
        auto release = [&]() -> void {
            while (!busy.empty() && busy.top().first <= now) {
                auto [t, i] = busy.top();
                busy.pop();
                idle.emplace(servers[i], i);
            }
        };

        // 依次处理每个任务
        for (int i = 0; i < tasks.size(); i++) {
            now = max(now, 1LL * i);
            release();
            if (idle.empty()) {
                now = busy.top().first; // 跳到 busy 的堆顶时刻，从而保证我们有空闲服务器
                release();
            }
            auto [w, j] = idle.top();
            idle.pop();
            busy.emplace(now + tasks[i], j);
            tasks[i] = j; // 直接把答案记在 tasks 中
        }
        return tasks;
    }
};
```

```go [sol-Go]
func assignTasks(servers, tasks []int) []int {
	// 初始化空闲服务器堆，保存 (服务器权重, 服务器下标)
	idle := make(hp, len(servers))
	for i, w := range servers {
		idle[i] = pair{w, i}
	}
	heap.Init(&idle)

	busy := hp{} // 繁忙服务器堆，保存 (服务器完成任务的时刻, 服务器下标)
	now := 0     // 当前时刻

	// 把已经完成任务的服务器从 busy 移至 idle
	release := func() {
		for len(busy) > 0 && busy[0].t <= now {
			i := heap.Pop(&busy).(pair).i
			heap.Push(&idle, pair{servers[i], i})
		}
	}

	// 依次处理每个任务
	for i, t := range tasks {
		now = max(now, i)
		release()
		if len(idle) == 0 {
			now = busy[0].t // 跳到 busy 的堆顶时刻，从而保证我们有空闲服务器
			release()
		}
		j := heap.Pop(&idle).(pair).i
		heap.Push(&busy, pair{now + t, j})
		tasks[i] = j // 直接把答案记在 tasks 中
	}
	return tasks
}

type pair struct{ t, i int }
type hp []pair

func (h hp) Len() int           { return len(h) }
func (h hp) Less(i, j int) bool { a, b := h[i], h[j]; return a.t < b.t || a.t == b.t && a.i < b.i }
func (h hp) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *hp) Push(v any)        { *h = append(*h, v.(pair)) }
func (h *hp) Pop() any          { a := *h; v := a[len(a)-1]; *h = a[:len(a)-1]; return v }
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n + m\log n)$ 或 $\mathcal{O}((n+m)\log n)$，其中 $n$ 是 $\textit{servers}$ 的长度，$m$ 是 $\textit{tasks}$ 的长度。虽然写了个二重循环，但每个任务至多入堆出堆一次，所以二重循环的循环次数为 $\mathcal{O}(m)$。由于有 $n$ 台服务器，所以堆的大小至多为 $n$，所以每次入堆出堆的时间复杂度为 $\mathcal{O}(\log n)$。初始化 $\textit{idle}$ 需要 $\mathcal{O}(n)$ 或 $\mathcal{O}(n\log n)$ 的时间，取决于是否使用了 $\mathcal{O}(n)$ 堆化的算法。
- 空间复杂度：$\mathcal{O}(n)$。
