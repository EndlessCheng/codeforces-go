定义一个括号的**深度**为其外面嵌套了多少层括号。最外层的括号深度为 $0$。

**题意**：删除所有深度为 $0$ 的括号，只保留深度大于 $0$ 的括号。

如何计算深度？遇到左括号就加一，遇到右括号就减一。

```py [sol-Python3]
class Solution:
    def removeOuterParentheses(self, s: str) -> str:
        ans = []
        depth = 0
        for ch in s:
            if ch == '(':
                if depth > 0:
                    ans.append(ch)
                depth += 1
            else:
                depth -= 1
                if depth > 0:
                    ans.append(ch)
        return ''.join(ans)
```

```java [sol-Java]
class Solution {
    public String removeOuterParentheses(String S) {
        char[] s = S.toCharArray();
        int size = 0;
        int depth = 0;
        for (char ch : s) {
            if (ch == '(') {
                if (depth > 0) {
                    s[size++] = ch;
                }
                depth++;
            } else {
                depth--;
                if (depth > 0) {
                    s[size++] = ch;
                }
            }
        }
        return new String(s, 0, size);
    }
}
```

char* removeOuterParentheses(char* s) {
    int size = 0;
    int depth = 0;

    for (int i = 0; s[i]; i++) {
        if (s[i] == '(') {
            if (depth > 0) {
                s[size++] = s[i];
            }
            depth++;
        } else {
            depth--;
            if (depth > 0) {
                s[size++] = s[i];
            }
        }
    }

    s[size] = '\0';
    return s;
}

func removeOuterParentheses(s string) string {

	depth := 0

		if ch == '(' {
			if depth > 0 {
				ans = append(ans, byte(ch))

			depth++
		} else {

			if depth > 0 {

			}
		}
	}
	return string(ans)

var removeOuterParentheses = function(s) {
    const ans = [];
    let depth = 0;
    for (const ch of s) {
        if (ch === '(') {
            if (depth > 0) {
                ans.push(ch);
            }
            depth++;
        } else {
            depth--;
            if (depth > 0) {
                ans.push(ch);
            }
        }
    }
    return ans.join('');
};

impl Solution {
    pub fn remove_outer_parentheses(s: String) -> String {
        let mut ans = vec![];
        let mut depth = 0;
        for ch in s.bytes() {
            if ch == b'(' {
                if depth > 0 {
                    ans.push(ch);
                }
                depth += 1;
            } else {
                depth -= 1;
                if depth > 0 {
                    ans.push(ch);
                }
            }
        }
        unsafe { String::from_utf8_unchecked(ans) }
    }
}

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n)$，其中 $n$ 是 $\textit{nums}$ 的长度。
- 空间复杂度：$\mathcal{O}(n)$ 或 $\mathcal{O}(1)$。如果能直接修改 $s$，可以做到 $\mathcal{O}(1)$ 空间，见 C 语言代码。
