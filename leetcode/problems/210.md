面对一堆杂乱的先修课约束，如何整理一份井井有条的课程学习顺序？

![拓扑排序](https://pic.leetcode.cn/1738131168-tWFNGZ-006-toposort.png)

首先，没有任何先修课的课程，可以直接学。如上图中的课程 $1$。

学完课程 $1$ 后，怎么知道还有哪些课可以学了？

我们可以记录每门课还剩下多少门先修课要学。在上图中，课程 $1$ 是课程 $2$ 和 $4$ 的先修课，学完课程 $1$ 后，课程 $2$ 和 $4$ 的剩余先修课减少 $1$。现在，课程 $4$ 没有要学的先修课了，所以接下来可以学课程 $4$。学完课程 $4$ 后，课程 $2,5,6$ 的剩余先修课减少 $1$，下一个要学的课程可以是 $2$ 也可以是 $5$。

一般地，把先修课关系表示成一张**有向图**。一门课的先修课个数，用图论术语来说，是这个节点的**入度**（in-degree）。

学完一门课后，邻居可能成为后续要学的课。这类似 BFS，我们可以用队列保存后续要学的课程顺序。为了得到答案，可以在学完课程（节点出队）时把课程的编号加入答案。

> **注**：由于我们只在入度恰好为 $0$ 时才会把节点入队，所以不会把节点重复入队，也不需要 $\textit{vis}$ 标记。

```py [sol-Python3]
class Solution:
    def findOrder(self, numCourses: int, prerequisites: list[list[int]]) -> list[int]:
        n = numCourses
        g = [[] for _ in range(n)]
        in_deg = [0] * n
        for a, b in prerequisites:
            g[b].append(a)
            in_deg[a] += 1  # 统计 a 的先修课数量

        # 没有先修课的课程，可以直接学
        q = deque(i for i, d in enumerate(in_deg) if d == 0)
        topo_order = []

        while q:
            x = q.popleft()  # 学习课程 x
            topo_order.append(x)
            for y in g[x]:  # 遍历 x 的邻居
                in_deg[y] -= 1  # 学完 x 后，y 的先修课数量减一
                if in_deg[y] == 0:  # y 的先修课全部学完
                    q.append(y)  # 加入学习队列

        if len(topo_order) < n:  # 图中有环
            return []
        return topo_order
```

```java [sol-Java]
class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        int n = numCourses;
        List<Integer>[] g = new ArrayList[n];
        Arrays.setAll(g, _ -> new ArrayList<>());
        int[] inDeg = new int[n];
        for (int[] pre : prerequisites) {
            int a = pre[0];
            int b = pre[1];
            g[b].add(a);
            inDeg[a]++; // 统计 a 的先修课数量
        }

        Queue<Integer> q = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            if (inDeg[i] == 0) { // 课程 i 没有先修课，可以直接学
                q.offer(i); // 加入学习队列
            }
        }

        int[] topoOrder = new int[n];
        int size = 0;
        while (!q.isEmpty()) {
            int x = q.poll(); // 学习课程 x
            topoOrder[size++] = x;
            for (int y : g[x]) { // 遍历 x 的邻居
                inDeg[y]--; // 学完 x 后，y 的先修课数量减一
                if (inDeg[y] == 0) { // y 的先修课全部学完
                    q.offer(y); // 加入学习队列
                }
            }
        }

        if (size < n) { // 图中有环
            return new int[0];
        }
        return topoOrder;
    }
}
```

```go [sol-Go]
func findOrder(numCourses int, prerequisites [][]int) []int {
	n := numCourses
	g := make([][]int, n)
	inDeg := make([]int, n)
	for _, pre := range prerequisites {
		a, b := pre[0], pre[1]
		g[b] = append(g[b], a)
		inDeg[a]++ // 统计 a 的先修课数量
	}

	q := make([]int, 0, n)
	topoOrder := q
	for i, d := range inDeg {
		if d == 0 { // 课程 i 没有先修课，可以直接学
			q = append(q, i) // 加入学习队列
		}
	}

	for len(q) > 0 {
		x := q[0] // 学习课程 x
		q = q[1:]
		for _, y := range g[x] { // 遍历 x 的邻居
			inDeg[y]-- // 学完 x 后，y 的先修课数量减一
			if inDeg[y] == 0 { // y 的先修课全部学完
				q = append(q, y) // 加入学习队列
			}
		}
	}

	if cap(q) > 0 { // 有节点没入队，说明图中有环
		return nil
	}
	return topoOrder[:n]
}
```

```js [sol-JavaScript]
var findOrder = function(numCourses, prerequisites) {
    const n = numCourses;
    const g = Array.from({ length: n }, () => []);
    const inDeg = Array(n).fill(0);
    for (const [a, b] of prerequisites) {
        g[b].push(a);
        inDeg[a]++; // 统计 a 的先修课数量
    }

    const q = new Queue(); // "@datastructures-js/queue"
    for (let i = 0; i < n; i++) {
        if (inDeg[i] === 0) { // 课程 i 没有先修课，可以直接学
            q.enqueue(i); // 加入学习队列
        }
    }

    const topoOrder = [];
    while (!q.isEmpty()) {
        const x = q.dequeue(); // 学习课程 x
        topoOrder.push(x);
        for (const y of g[x]) { // 遍历 x 的邻居
            inDeg[y]--; // 学完 x 后，y 的先修课数量减一
            if (inDeg[y] === 0) { // y 的先修课全部学完
                q.enqueue(y); // 加入学习队列
            }
        }
    }

    if (topoOrder.length < n) { // 图中有环
        return [];
    }
    return topoOrder;
};
```

```rust [sol-Rust]
use std::collections::VecDeque;

impl Solution {
    pub fn find_order(num_courses: i32, prerequisites: Vec<Vec<i32>>) -> Vec<i32> {
        let n = num_courses as usize;
        let mut g = vec![vec![]; n];
        let mut in_deg = vec![0; n];
        for pre in prerequisites {
            let a = pre[0] as usize;
            let b = pre[1] as usize;
            g[b].push(a);
            in_deg[a] += 1; // 统计 a 的先修课数量
        }

        let mut q = VecDeque::new();
        for (i, &d) in in_deg.iter().enumerate() {
            if d == 0 { // 课程 i 没有先修课，可以直接学
                q.push_back(i); // 加入学习队列
            }
        }

        let mut topo_order = vec![];
        while let Some(x) = q.pop_front() {
            topo_order.push(x as i32); // 学习课程 x
            for &y in &g[x] { // 遍历 x 的邻居
                in_deg[y] -= 1; // 学完 x 后，y 的先修课数量减一
                if in_deg[y] == 0 { // y 的先修课全部学完
                    q.push_back(y); // 加入学习队列
                }
            }
        }

        if topo_order.len() < n { // 图中有环
            return vec![];
        }
        topo_order
    }
}
```

#### 复杂度分析

- 时间复杂度：$\mathcal{O}(n+m)$，其中 $n = \textit{numCourses}$，$m$ 是 $\textit{prerequisites}$ 的长度。
- 空间复杂度：$\mathcal{O}(n+m)$。
