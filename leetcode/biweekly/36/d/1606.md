例如 $k=4$，当我们处理请求 $i=2$ 时，要按照 $2,3,0,1$ 的环状顺序寻找空闲服务器的编号。这个顺序不是连续的，不好用最小堆维护。

有没有一种办法，可以把环「拉直」呢？

为了能用最小堆维护空闲服务器的编号，当我们处理请求 $i=2$ 时，如果此时服务器 $0$ 和 $1$ 变成空闲的，我们可以把编号 $0$ 视作模 $k=4$ [同余](https://leetcode.cn/discuss/post/3584387/fen-xiang-gun-mo-yun-suan-de-shi-jie-dan-7xgu/) 的虚拟编号 $4$，把编号 $1$ 视作模 $k=4$ 同余的虚拟编号 $5$，那么服务器编号就从原来的环状 $2,3,0,1$ 拉直成 $2,3,4,5$。这样既能保证 $2$ 是最小的编号，又能通过模 $k$ 把虚拟编号还原成对应的服务器编号。

一般地，在把服务器编号从繁忙服务器堆移至空闲服务器堆的过程中，计算服务器 $j$ 的虚拟编号 $\textit{virtual}$，满足 $\textit{virtual} \equiv j \pmod k$ 且 $\textit{virtual}\in [i,i+k)$。由于 $\textit{virtual} \equiv j \equiv i+(j-i) \pmod k$，结合 $\textit{virtual}\in [i,i+k)$，得

$$
\textit{virtual} = i + (j-i)\bmod k
$$

> 注：部分语言计算 $(j-i)\bmod k$ 可能得到负数，需要调整到 $[0,k)$ 中。

#### 答疑

**问**：处理请求 $i$ 时，$\textit{idle}$ 中的虚拟编号都 $\ge i$ 吗？

**答**：是的。这可以用数学归纳法证明。$i=0$ 时显然成立。假设处理请求 $i=m$ 时，$\textit{idle}$ 中的虚拟编号都 $\ge m$。处理请求 $i=m$ 时，首先添加到 $\textit{idle}$ 中的虚拟编号都 $\ge i = m$，然后我们会从 $\textit{idle}$ 中取出最小值，这个最小值要么是 $m$，要么大于 $m$，无论是哪种情况，取出最小值后，剩余虚拟编号都 $\ge m+1$。所以处理请求 $i=m+1$ 时，$\textit{idle}$ 中的虚拟编号都 $\ge m+1$。综上，由数学归纳法可知，原命题成立。

```py [sol-Python3]
class Solution:
    def busiestServers(self, k: int, arrival: list[int], load: list[int]) -> list[int]:
        # 小优化：如果 k >= len(arrival)，我们只会用到前 len(arrival) 台服务器，且每台服务器恰好使用一次
        if k >= len(arrival):
            return list(range(len(arrival)))

        idle = list(range(k))  # 空闲服务器堆，保存空闲服务器的虚拟编号
        busy = []  # 繁忙服务器堆，保存 (服务器完成请求的时刻, 服务器编号)
        cnt = [0] * k  # cnt[i] 统计服务器 i 处理的请求数
        max_cnt = 0
        ans = []

        for i, (t, d) in enumerate(zip(arrival, load)):
            # 把已经完成请求的服务器从 busy 移至 idle
            while busy and busy[0][0] <= t:
                _, j = heappop(busy)
                # 把服务器 j 在 [i,i+k) 中的同余值作为该服务器的虚拟编号
                heappush(idle, i + (j - i) % k)

            # 没有空闲服务器，舍弃该请求
            if not idle:
                continue

            j = heappop(idle) % k
            heappush(busy, (t + d, j))

            cnt[j] += 1
            if cnt[j] > max_cnt:  # j 是最繁忙的服务器
                max_cnt = cnt[j]
                ans = [j]
            elif cnt[j] == max_cnt:  # j 是最繁忙的服务器之一
                ans.append(j)

        return ans
```

```java [sol-Java]
class Solution {
    public List<Integer> busiestServers(int k, int[] arrival, int[] load) {
        // 小优化：如果 k >= n，我们只会用到前 n 台服务器，且每台服务器恰好使用一次
        int n = arrival.length;
        if (k >= n) {
            List<Integer> ans = new ArrayList<>(n);
            for (int i = 0; i < n; i++) {
                ans.add(i);
            }
            return ans;
        }

        // 初始化空闲服务器堆，保存空闲服务器的虚拟编号
        PriorityQueue<Integer> idle = new PriorityQueue<>();
        for (int i = 0; i < k; i++) {
            idle.offer(i);
        }

        // 繁忙服务器堆，保存 (服务器完成请求的时刻, 服务器编号)
        PriorityQueue<int[]> busy = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        int[] cnt = new int[k]; // cnt[i] 统计服务器 i 处理的请求数
        int maxCnt = 0;
        List<Integer> ans = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            // 把已经完成请求的服务器从 busy 移至 idle
            while (!busy.isEmpty() && busy.peek()[0] <= arrival[i]) {
                int j = busy.poll()[1];
                // 把服务器 j 在 [i,i+k) 中的同余值 virtual 作为该服务器的虚拟编号
                int virtual = i + (j - i % k + k) % k; // +k 保证取模结果非负
                idle.offer(virtual);
            }

            // 没有空闲服务器，舍弃该请求
            if (idle.isEmpty()) {
                continue;
            }

            int j = idle.poll() % k;
            busy.offer(new int[]{arrival[i] + load[i], j});

            cnt[j]++;
            if (cnt[j] > maxCnt) { // j 是最繁忙的服务器
                maxCnt = cnt[j];
                ans.clear();
                ans.add(j);
            } else if (cnt[j] == maxCnt) { // j 是最繁忙的服务器之一
                ans.add(j);
            }
        }

        return ans;
    }
}
```

```cpp [sol-C++]
class Solution {
public:
    vector<int> busiestServers(int k, vector<int>& arrival, vector<int>& load) {
        // 小优化：如果 k >= n，我们只会用到前 n 台服务器，且每台服务器恰好使用一次
        int n = arrival.size();
        if (k >= n) {
            vector<int> ans(n);
            ranges::iota(ans, 0);
            return ans;
        }

        // 初始化空闲服务器堆，保存空闲服务器的虚拟编号
        priority_queue<int, vector<int>, greater<>> idle;
        for (int i = 0; i < k; i++) {
            idle.push(i);
        }

        // 繁忙服务器堆，保存 (服务器完成请求的时刻, 服务器编号)
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> busy; 
        vector<int> cnt(k); // cnt[i] 统计服务器 i 处理的请求数
        int max_cnt = 0;
        vector<int> ans;

        for (int i = 0; i < n; i++) {
            // 把已经完成请求的服务器从 busy 移至 idle
            while (!busy.empty() && busy.top().first <= arrival[i]) {
                auto [_, j] = busy.top();
                busy.pop();
                // 把服务器 j 在 [i,i+k) 中的同余值 virtual_id 作为该服务器的虚拟编号
                int virtual_id = i + (j - i % k + k) % k; // +k 保证取模结果非负
                idle.push(virtual_id);
            }

            // 没有空闲服务器，舍弃该请求
            if (idle.empty()) {
                continue;
            }

            int j = idle.top() % k;
            idle.pop();
            busy.emplace(arrival[i] + load[i], j);

            cnt[j]++;
            if (cnt[j] > max_cnt) { // j 是最繁忙的服务器
                max_cnt = cnt[j];
                ans = {j};
            } else if (cnt[j] == max_cnt) { // j 是最繁忙的服务器之一
                ans.push_back(j);
            }
        }

        return ans;
    }
};
```

```go [sol-Go]
func busiestServers(k int, arrival, load []int) (ans []int) {
	// 小优化：如果 k >= len(arrival)，我们只会用到前 len(arrival) 台服务器，且每台服务器恰好使用一次
	if k >= len(arrival) {
		ans = make([]int, len(arrival))
		for i := range ans {
			ans[i] = i
		}
		return
	}

	// 初始化空闲服务器堆，保存空闲服务器的虚拟编号
	idle := hp{make([]int, k)}
	for i := range k {
		idle.IntSlice[i] = i
	}

	busy := hp2{} // 繁忙服务器堆，保存 (服务器完成请求的时刻, 服务器编号)
	cnt := make([]int, k) // cnt[i] 统计服务器 i 处理的请求数
	maxCnt := 0

	for i, t := range arrival {
		// 把已经完成请求的服务器从 busy 移至 idle
		for len(busy) > 0 && busy[0].t <= t {
			j := heap.Pop(&busy).(pair).i
			// 把服务器 j 在 [i,i+k) 中的同余值 virtual 作为该服务器的虚拟编号
			virtual := i + (j-i%k+k)%k // +k 保证取模结果非负
			heap.Push(&idle, virtual)
		}

		// 没有空闲服务器，舍弃该请求
		if idle.Len() == 0 {
			continue
		}

		j := heap.Pop(&idle).(int) % k
		heap.Push(&busy, pair{t + load[i], j})

		cnt[j]++
		if cnt[j] > maxCnt { // j 是最繁忙的服务器
			maxCnt = cnt[j]
			ans = []int{j}
		} else if cnt[j] == maxCnt { // j 是最繁忙的服务器之一
			ans = append(ans, j)
		}
	}

	return
}

type hp struct{ sort.IntSlice }
func (h *hp) Push(v any) { h.IntSlice = append(h.IntSlice, v.(int)) }
func (h *hp) Pop() any   { a := h.IntSlice; v := a[len(a)-1]; h.IntSlice = a[:len(a)-1]; return v }

type pair struct{ t, i int }
type hp2 []pair
func (h hp2) Len() int           { return len(h) }
func (h hp2) Less(i, j int) bool { return h[i].t < h[j].t }
func (h hp2) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *hp2) Push(v any)        { *h = append(*h, v.(pair)) }
func (h *hp2) Pop() any          { a := *h; v := a[len(a)-1]; *h = a[:len(a)-1]; return v }
```

#### 复杂度分析

设 $n$ 是 $\textit{arrival}$ 的长度。

- 时间复杂度：
    - 如果 $k\ge n$：$\mathcal{O}(n)$。
    - 如果 $k < n$：$\mathcal{O}(n\log k)$。虽然写了个二重循环，但每个请求至多入堆出堆一次，所以二重循环的循环次数为 $\mathcal{O}(n)$。由于堆的大小不超过 $k$，所以每次入堆出堆的时间复杂度为 $\mathcal{O}(\log k)$。
- 空间复杂度：
    - 如果 $k\ge n$：$\mathcal{O}(1)$。返回值不计入。
    - 如果 $k < n$：$\mathcal{O}(k)$。
