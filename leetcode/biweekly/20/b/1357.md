class Cashier:
    def __init__(self, n: int, discount: int, products: list[int], prices: list[int]):
        self.product_to_price = dict(zip(products, prices))
        self.customers = 0  # 顾客个数
        self.n = n
        self.discount = 1 - discount / 100

    def getBill(self, product: list[int], amount: list[int]) -> float:
        ans = sum(self.product_to_price[p] * a for p, a in zip(product, amount))
        self.customers += 1
        if self.customers % self.n == 0:  # 第 n,2n,3n,... 个顾客会得到 discount 的折扣
            ans *= self.discount
        return ans
